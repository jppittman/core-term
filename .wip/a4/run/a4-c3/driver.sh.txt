#!/usr/bin/env bash
# A4 closing gates round 3 driver (TAG=a4-c3). Sequential: the gate scripts edit tests.rs for one build.
set -u
REPO=/home/user/pixelflow
OUT=$REPO/.wip/a4/run/a4-c3
G=$REPO/.wip/gates
TAG=a4-c3
cd "$REPO" || exit 9
prog() { echo "$1" >> "$OUT/progress.txt"; }
avail_mb() { df -BM --output=avail / | tail -1 | tr -dc 0-9; }
CLEANED=0
guard() {
  local a; a=$(avail_mb)
  if [ "$a" -lt 3072 ] && [ "$CLEANED" = 0 ]; then
    CLEANED=1; prog "avail ${a}M < 3 GB: cargo clean -q --profile dev -p pixelflow-pipeline"
    cargo clean -q --profile dev -p pixelflow-pipeline >> "$OUT/clean.txt" 2>&1
    a=$(avail_mb)
  fi
  if [ "$a" -lt 2048 ]; then prog "STOP: ${a}M free (< 2 GB)"; exit 3; fi
}
step() { # NAME CMD...  (stdout+stderr to $OUT/NAME.txt)
  local name=$1; shift; guard
  "$@" > "$OUT/$name.txt" 2>&1; local rc=$?
  prog "$name done: EXIT $rc (avail $(avail_mb)M)"
  return $rc
}

: > "$OUT/progress.txt"
prog "start $(date -u +%FT%TZ) HEAD $(git rev-parse --short HEAD) avail $(avail_mb)M"

# Gates 1-3, 5
step byteprobe.stdout bash $G/byteprobe.sh $TAG
step gate1b.stdout bash $G/gate1b.sh $TAG
step gate2.stdout bash $G/gate2.sh $TAG
step gate2x.stdout bash $G/gate2x.sh $TAG
guard
bash $G/measure.sh $TAG > "$OUT/measure.stdout.txt" 2> "$OUT/measure.stderr.txt"; prog "measure done: EXIT $? (avail $(avail_mb)M)"

# cargo tests
step test.pixelflow-ir cargo test -p pixelflow-ir
step test.pixelflow-search-default cargo test -p pixelflow-search
step test.pixelflow-search-all cargo test -p pixelflow-search --all-features
step test.pixelflow-search-telemetry cargo test -p pixelflow-search --features saturation-telemetry
PIXELFLOW_ISA=avx2 step test.pixelflow-codegen-avx2 cargo test -p pixelflow-codegen
PIXELFLOW_ISA=avx512 step test.pixelflow-codegen-avx512 cargo test -p pixelflow-codegen
step test.pixelflow-compiler cargo test -p pixelflow-compiler
step test.pixelflow-graphics cargo test -p pixelflow-graphics
step test.pixelflow-pipeline cargo test -p pixelflow-pipeline
step test.pixelflow-core cargo test -p pixelflow-core
step test.pixelflow-runtime cargo test -p pixelflow-runtime
prog "TESTS-DONE"

# clippy (1.99), fmt
CARGO_TARGET_DIR=/home/user/scratch/a3-close1/t199 step clippy cargo +1.99.0 clippy --workspace --all-targets --all-features -- -D warnings
step fmt cargo fmt --all -- --check

# no-default-features, docs (check-only target dir on tmpfs)
export TCHK=/dev/shm/a4-c2-tchk
RUSTFLAGS="-D warnings" CARGO_TARGET_DIR=$TCHK step nodef-pixelflow-ir cargo check -p pixelflow-ir --no-default-features
RUSTFLAGS="-D warnings" CARGO_TARGET_DIR=$TCHK step nodef-pixelflow-search cargo check -p pixelflow-search --no-default-features
RUSTDOCFLAGS="-D warnings" CARGO_TARGET_DIR=$TCHK step doc-codegen cargo doc -p pixelflow-codegen --no-deps --document-private-items
for c in pixelflow-ir pixelflow-search pixelflow-codegen; do
  guard
  CARGO_TARGET_DIR=$TCHK cargo doc -p $c --no-deps --document-private-items > "$OUT/rustdoc.$c.raw.txt" 2>&1
  python3 -I $REPO/.wip/a4/rustdoc_norm.py < "$OUT/rustdoc.$c.raw.txt" > "$OUT/rustdoc.$c.norm.txt"
  python3 -I $REPO/.wip/a4/rustdoc_norm.py --compare $REPO/.wip/a4/rustdoc-base.$c.txt < "$OUT/rustdoc.$c.raw.txt" > "$OUT/rustdoc.$c.compare.txt" 2>&1
  prog "rustdoc $c compare EXIT $?"
done

# scripts
for s in check-intent check-doc-paths check-cfg-encapsulation check-bin-declarations check-emit-boundary; do
  step $s bash scripts/$s.sh
done
PATH=/home/user/scratch/cargo-hack-root/bin:$PATH CARGO_TARGET_DIR=$TCHK step check-feature-matrix bash scripts/check-feature-matrix.sh

# greps
grep -rnwE 'A4' --include=*.rs pixelflow-* > "$OUT/grep-A4.txt" 2>&1; prog "grep-A4 hits: $(wc -l < "$OUT/grep-A4.txt")"
grep -rln '\.wip' --include=*.rs --include=*.md . 2>/dev/null | grep -v '^./.wip/' > "$OUT/grep-wip.txt"; prog "grep-wip hits: $(wc -l < "$OUT/grep-wip.txt")"
prog "ALLDONE $(date -u +%FT%TZ)"
