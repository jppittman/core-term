// Scratch probe: the value_bounds_jit traffic pins' kernels, counted by the NEON backend
// (this scratch copy of pixelflow-codegen reports NEON's traffic beside the host's code).
#![cfg(target_arch = "x86_64")]
use pixelflow_codegen::emit::{self, EmitTraffic};
use pixelflow_ir::fold::Monoid;
use pixelflow_ir::{Kernel, LatticeShape, Uniform};

const SHAPE: LatticeShape = LatticeShape::new([64, 4]);
const WALL: u32 = 48;

fn count(v: f32) -> Kernel { Uniform::new(v).kernel() }
fn zero() -> Kernel { Kernel::constant(0.0) }
fn wall(k: &Kernel) -> Kernel {
    let terms: Vec<Kernel> = (0..WALL).map(|m| Kernel::x().add(k).mul(&Kernel::constant((m + 1) as f32))).collect();
    let chain = |t: &[Kernel]| t[1..].iter().fold(t[0].clone(), |a, x| a.add(x));
    let rev: Vec<Kernel> = terms.iter().rev().cloned().collect();
    chain(&terms).add(&chain(&rev))
}
fn counted(k: &Kernel) -> (EmitTraffic, u64) {
    let (a, root) = k.parts();
    let r = emit::compile(a, root, SHAPE).expect("compiles");
    (r.traffic, r.frame_bytes)
}
fn show(name: &str, v: &(EmitTraffic, u64), l: &(EmitTraffic, u64)) {
    eprintln!("NEON {name}: value trips={:?} frame={}", v.0.trips, v.1);
    eprintln!("NEON {name}: literal trips={:?} frame={}", l.0.trips, l.1);
    for (s, (a, b)) in v.0.scopes.iter().zip(l.0.scopes.iter()).enumerate() {
        eprintln!("NEON {name}: scope {s}: value instr={} loads={} stores={} | literal instr={} loads={} stores={}", a.instructions, a.loads, a.stores, b.instructions, b.loads, b.stores);
    }
}
#[test]
fn neon_counts() {
    let v = counted(&Kernel::over_values(Monoid::SUM, &zero(), &count(7.0), wall));
    let l = counted(&Kernel::sum_over(7, wall));
    show("pin1 wall", &v, &l);
    let v = counted(&Kernel::x().mul(&Kernel::over_values(Monoid::SUM, &zero(), &count(7.0), |k| Kernel::x().add(k))));
    let l = counted(&Kernel::x().mul(&Kernel::sum_over(7, |k| Kernel::x().add(k))));
    show("pin2 guard-once", &v, &l);
    let v = counted(&Kernel::y().mul(&Kernel::over_values(Monoid::SUM, &zero(), &count(7.0), |k| Kernel::y().add(k))));
    let l = counted(&Kernel::y().mul(&Kernel::sum_over(7, |k| Kernel::y().add(k))));
    show("pin3 y-only", &v, &l);
    let v = counted(&Kernel::sum_over(8, |i| Kernel::over_values(Monoid::SUM, &zero(), &count(8.0), |j| Kernel::x().add(j).add(i))));
    let l = counted(&Kernel::sum_over(8, |i| Kernel::sum_over(8, |j| Kernel::x().add(j).add(i))));
    show("pin4 enclosing", &v, &l);
}
