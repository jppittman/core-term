//! Round-4 run-time probes of value-bounded folds (review scratch, not for the tree).

use std::sync::mpsc::{self, RecvTimeoutError};
use std::time::Duration;

use pixelflow_codegen::emit;
use pixelflow_codegen::jit_cache;
use pixelflow_ir::fold::{Binder, Bound, Fold, Monoid};
use pixelflow_ir::{ExprArena, ExprId, Kernel, LatticeShape, OpKind, Uniform};

const SHAPE: LatticeShape = LatticeShape::new([64, 4]);
const RAGGED: LatticeShape = LatticeShape::new([37, 3]);
const ORIGIN: [f32; 2] = [1.0, 2.0];
const HANG: Duration = Duration::from_secs(60);
const TOP: f32 = 16_777_216.0;

fn zero() -> Kernel {
    Kernel::constant(0.0)
}

fn within<T: Send + 'static>(f: impl FnOnce() -> T + Send + 'static) -> T {
    let (tx, rx) = mpsc::channel();
    let worker = std::thread::spawn(move || {
        let _ = tx.send(f());
    });
    match rx.recv_timeout(HANG) {
        Ok(value) => value,
        Err(RecvTimeoutError::Timeout) => panic!("HANG"),
        Err(RecvTimeoutError::Disconnected) => match worker.join() {
            Err(panic) => std::panic::resume_unwind(panic),
            Ok(()) => unreachable!(),
        },
    }
}

fn call(code: &emit::CompiledKernel, block: &[f32], shape: LatticeShape) -> Vec<f32> {
    let [width, height] = shape.extent().map(|n| n as usize);
    let mut out = vec![f32::from_bits(0x7fc0_1234); width * height];
    let ctx: [*const f32; 2] = [block.as_ptr(), ORIGIN.as_ptr()];
    unsafe { code.call(ctx.as_ptr(), out.as_mut_ptr(), width) };
    out
}

fn run_arena(arena: &ExprArena, root: ExprId, shape: LatticeShape, block: &[f32]) -> Vec<f32> {
    let compiled = emit::compile(arena, root, shape).expect("compiles");
    call(&compiled.code, block, shape)
}

fn run_raw(kernel: &Kernel, shape: LatticeShape) -> Vec<f32> {
    let (arena, root) = kernel.parts();
    let block: Vec<f32> = arena.uniforms().iter().map(|d| d.default).collect();
    run_arena(arena, root, shape, &block)
}

fn run_linked(kernel: &Kernel, shape: LatticeShape) -> Vec<f32> {
    let linked = jit_cache::compile(kernel, shape).expect("compiles");
    let block: Vec<f32> = linked.uniforms.iter().map(|d| d.default).collect();
    call(&linked.kernel, &block, shape)
}

fn at(shape: LatticeShape, i: usize) -> (f32, f32) {
    let width = shape.extent()[0] as usize;
    (ORIGIN[0] + (i % width) as f32, ORIGIN[1] + (i / width) as f32)
}

fn both(
    shape: LatticeShape,
    build: impl Fn() -> Kernel + Send + Copy + 'static,
) -> [(&'static str, Vec<f32>); 2] {
    [
        ("raw", within(move || run_raw(&build(), shape))),
        ("linked", within(move || run_linked(&build(), shape))),
    ]
}

#[track_caller]
fn check<const P: usize>(
    shape: LatticeShape,
    planes: [(&'static str, Vec<f32>); P],
    ctx: &str,
    want: impl Fn(f32, f32) -> f32,
) {
    for (path, plane) in planes {
        for (i, &got) in plane.iter().enumerate() {
            let (x, y) = at(shape, i);
            let w = want(x, y);
            let same = if w.is_nan() { got.is_nan() } else { got.to_bits() == w.to_bits() };
            assert!(same, "{ctx} ({path}) at ({x},{y}): got {got} want {w}");
        }
    }
}

fn fold_of(monoid: Monoid, lo: u64, hi: u64, term: impl Fn(f32) -> f32) -> f32 {
    let identity = f32::from_bits(monoid.identity().bits());
    let combine = |a: f32, b: f32| match monoid {
        Monoid::SUM => a + b,
        Monoid::PRODUCT => a * b,
        Monoid::MIN => a.min(b),
        Monoid::MAX => a.max(b),
        _ => unreachable!(),
    };
    (lo..hi).fold(identity, |acc, k| combine(acc, term(k as f32)))
}

fn in_domain(v: f32) -> bool {
    (0.0..=TOP).contains(&v) && v.floor() == v
}

/// `[lo, hi)` as the fold's index set, `hi` clamped below by `lo`.
fn span(lo: f32, hi: f32) -> (u64, u64) {
    let (l, h) = (lo as u64, hi as u64);
    (l, h.max(l))
}

/// One uniform is the outer loop's end (read at its latch) and the inner
/// loop's seed: Σ_{i<u} Σ_{j∈[u,8)} (X + i + j).
#[test]
fn p1_outer_end_is_inner_seed() {
    for n in [0.0f32, 1.0, 3.0, 7.0, 8.0, 10.0, f32::NAN] {
        let planes = both(SHAPE, move || {
            let u = Uniform::new(n);
            Kernel::over_values(Monoid::SUM, &zero(), &u.kernel(), move |i| {
                Kernel::over_values(Monoid::SUM, &u.kernel(), &Kernel::constant(8.0), |j| {
                    Kernel::x().add(i).add(j)
                })
            })
        });
        check(SHAPE, planes, &format!("p1 n={n}"), |x, _| {
            if !in_domain(n) {
                return f32::NAN;
            }
            fold_of(Monoid::SUM, 0, n as u64, |i| {
                let (l, h) = span(n, 8.0);
                fold_of(Monoid::SUM, l, h, |j| x + i + j)
            })
        });
    }
}

/// A wall of values inside, with one uniform the outer end and the inner seed.
fn wall(k: &Kernel, terms_n: u32) -> Kernel {
    let terms: Vec<Kernel> = (0..terms_n)
        .map(|m| Kernel::x().add(k).mul(&Kernel::constant((m + 1) as f32)))
        .collect();
    let chain = |terms: &[Kernel]| terms[1..].iter().fold(terms[0].clone(), |acc, t| acc.add(t));
    let reversed: Vec<Kernel> = terms.iter().rev().cloned().collect();
    chain(&terms).add(&chain(&reversed))
}

#[test]
fn p6_outer_end_is_inner_seed_under_pressure() {
    for n in [0.0f32, 2.0, 5.0, f32::NAN] {
        let planes = both(RAGGED, move || {
            let u = Uniform::new(n);
            Kernel::over_values(Monoid::SUM, &zero(), &u.kernel(), move |i| {
                Kernel::over_values(Monoid::SUM, &u.kernel(), &Kernel::constant(6.0), |j| {
                    wall(&j.add(i), 40)
                })
            })
        });
        let w: f32 = (1..=40).map(|m| m as f32).sum();
        check(RAGGED, planes, &format!("p6 n={n}"), |x, _| {
            if !in_domain(n) {
                return f32::NAN;
            }
            fold_of(Monoid::SUM, 0, n as u64, |i| {
                let (l, h) = span(n, 6.0);
                fold_of(Monoid::SUM, l, h, |j| 2.0 * w * (x + (j + i)))
            })
        });
    }
}

/// Three levels: the middle counter seeds the innermost, the outermost end
/// and the innermost end are the one uniform.
/// Σ_{i<u} Σ_{j∈[i,8)} Σ_{k∈[j,u)} (X + k).
#[test]
fn p7_three_levels_share_one_end() {
    for n in [0.0f32, 3.0, 6.0, 9.0, f32::NAN] {
        let planes = both(SHAPE, move || {
            let u = Uniform::new(n);
            Kernel::over_values(Monoid::SUM, &zero(), &u.kernel(), move |i| {
                Kernel::over_values(Monoid::SUM, i, &Kernel::constant(8.0), move |j| {
                    Kernel::over_values(Monoid::SUM, j, &u.kernel(), |k| Kernel::x().add(k))
                })
            })
        });
        check(SHAPE, planes, &format!("p7 n={n}"), |x, _| {
            if !in_domain(n) {
                return f32::NAN;
            }
            fold_of(Monoid::SUM, 0, n as u64, |i| {
                let (a, b) = span(i, 8.0);
                fold_of(Monoid::SUM, a, b, |j| {
                    let (c, d) = span(j, n);
                    fold_of(Monoid::SUM, c, d, |k| x + k)
                })
            })
        });
    }
}

/// Lo values out of the domain for both monoid kinds, under a literal outer
/// loop whose index is read in the body.
#[test]
fn p8_value_lo_classes_inside_a_literal_loop() {
    let classes = [
        0.0f32, -0.0, 1.0, 7.0, TOP, f32::NAN, f32::INFINITY, f32::NEG_INFINITY, -1.0, 2.5, TOP + 2.0,
    ];
    for lo in classes {
        let planes = both(SHAPE, move || {
            let u = Uniform::new(lo);
            Kernel::sum_over(3, move |i| {
                Kernel::over_values(Monoid::MAX, &u.kernel(), &Kernel::constant(9.0), |k| {
                    Kernel::x().add(k).add(i)
                })
            })
        });
        check(SHAPE, planes, &format!("p8 lo={lo}"), |x, _| {
            if !in_domain(lo) {
                return f32::NAN;
            }
            let (l, h) = span(lo, 9.0);
            fold_of(Monoid::SUM, 0, 3, |i| fold_of(Monoid::MAX, l, h, |k| x + k + i))
        });
    }
}

/// A hand arena where the bound is the uniform halved: a fraction for odd u.
#[test]
fn p9_a_computed_fraction_bound() {
    for (u, want) in [(6.0f32, Some(3u64)), (7.0, None), (0.0, Some(0)), (-0.0, Some(0)), (-2.0, None)] {
        let mut a = ExprArena::new();
        let slot = a.declare_uniform(Uniform::new(u).decl());
        let uni = a.push_uniform(slot);
        let half = a.push_const(0.5);
        let end = a.push_binary(OpKind::Mul, uni, half);
        let binder = Binder::from_slot(0).expect("slot");
        let (x, k) = (a.push_var(0), a.push_var(binder.var()));
        let body = a.push_binary(OpKind::Add, x, k);
        let root = a.push_reduce(Fold::new(Monoid::SUM, binder), Bound::Lit(0), Bound::Value(end), body);
        let plane = within(move || run_arena(&a, root, SHAPE, &[u]));
        check(SHAPE, [("hand", plane)], &format!("p9 u={u}"), |x, _| match want {
            Some(n) => fold_of(Monoid::SUM, 0, n, |k| x + k),
            None => f32::NAN,
        });
    }
}

/// An inner end that is the outer counter, under a wall of values.
/// Σ_{i<8} Σ_{j<i} wall(X + j).
#[test]
fn p10_inner_end_is_outer_counter_under_pressure() {
    let planes = both(RAGGED, || {
        Kernel::sum_over(8, |i| Kernel::over_values(Monoid::SUM, &zero(), i, |j| wall(j, 40)))
    });
    let w: f32 = (1..=40).map(|m| m as f32).sum();
    check(RAGGED, planes, "p10", |x, _| {
        fold_of(Monoid::SUM, 0, 8, |i| fold_of(Monoid::SUM, 0, i as u64, |j| 2.0 * w * (x + j)))
    });
}

/// `Σ_m (X + k)·(m + 1)` twice over, pushed by hand.
fn push_wall(a: &mut ExprArena, k: ExprId, n: u32) -> ExprId {
    let x = a.push_var(0);
    let xk = a.push_binary(OpKind::Add, x, k);
    let terms: Vec<ExprId> = (0..n)
        .map(|m| {
            let c = a.push_const((m + 1) as f32);
            a.push_binary(OpKind::Mul, xk, c)
        })
        .collect();
    let mut up = terms[0];
    for &t in &terms[1..] {
        up = a.push_binary(OpKind::Add, up, t);
    }
    let mut down = terms[n as usize - 1];
    for &t in terms.iter().rev().skip(1) {
        down = a.push_binary(OpKind::Add, down, t);
    }
    a.push_binary(OpKind::Add, up, down)
}

/// Computed bounds from the outer counter, under pressure, by hand:
/// Σ_{i<6} Σ_{j∈[i+1, i+u)} wall(j), u a uniform.
#[test]
fn p11_computed_bounds_from_the_outer_counter_under_pressure() {
    for u in [3.0f32, 1.0, 0.0, 2.5, f32::NAN] {
        let mut a = ExprArena::new();
        let slot = a.declare_uniform(Uniform::new(u).decl());
        let uni = a.push_uniform(slot);
        let outer = Binder::from_slot(1).expect("slot");
        let inner = Binder::from_slot(0).expect("slot");
        let i = a.push_var(outer.var());
        let one = a.push_const(1.0);
        let lo = a.push_binary(OpKind::Add, i, one);
        let hi = a.push_binary(OpKind::Add, i, uni);
        let j = a.push_var(inner.var());
        let body = push_wall(&mut a, j, 40);
        let inner_fold = a.push_reduce(Fold::new(Monoid::SUM, inner), Bound::Value(lo), Bound::Value(hi), body);
        let root = a.push_reduce(Fold::new(Monoid::SUM, outer), Bound::Lit(0), Bound::Lit(6), inner_fold);
        let plane = within(move || run_arena(&a, root, RAGGED, &[u]));
        let w: f32 = (1..=40).map(|m| m as f32).sum();
        check(RAGGED, [("hand", plane)], &format!("p11 u={u}"), |x, _| {
            let mut acc = 0.0f32;
            for i in 0..6u64 {
                let (lo, hi) = (i as f32 + 1.0, i as f32 + u);
                if !in_domain(lo) || !in_domain(hi) {
                    return f32::NAN;
                }
                let (l, h) = span(lo, hi);
                acc += fold_of(Monoid::SUM, l, h, |j| 2.0 * w * (x + j));
            }
            acc
        });
    }
}

/// A value-bounded fold that reads a buffer at its index: Σ_{k<u} buf[k, 0],
/// the buffer 4 wide, u up to 1000. The read is clamped (expand_gather).
#[test]
fn p12_a_buffer_read_at_a_value_bounded_index() {
    use pixelflow_ir::arena::{BufferDecl, BufferIdentity};
    for u in [0.0f32, 3.0, 4.0, 1000.0, f32::NAN] {
        let mut a = ExprArena::new();
        let b = a.declare_buffer(BufferDecl { id: BufferIdentity::mint(), width: 4, height: 1 });
        let buf = a.push_buffer(b);
        let slot = a.declare_uniform(Uniform::new(u).decl());
        let uni = a.push_uniform(slot);
        let binder = Binder::from_slot(0).expect("slot");
        let k = a.push_var(binder.var());
        let zero = a.push_const(0.0);
        let g = a.push_ternary(OpKind::Gather, buf, k, zero);
        let root = a.push_reduce(Fold::new(Monoid::SUM, binder), Bound::Lit(0), Bound::Value(uni), g);
        let plane = within(move || {
            let compiled = emit::compile(&a, root, SHAPE).expect("compiles");
            let data: Vec<f32> = vec![1.0, 10.0, 100.0, 1000.0];
            let block = [u];
            let [width, height] = SHAPE.extent().map(|n| n as usize);
            let mut out = vec![0.0f32; width * height];
            let ctx: [*const f32; 3] = [data.as_ptr(), block.as_ptr(), ORIGIN.as_ptr()];
            unsafe { compiled.code.call(ctx.as_ptr(), out.as_mut_ptr(), width) };
            out
        });
        let data = [1.0f32, 10.0, 100.0, 1000.0];
        check(SHAPE, [("hand", plane)], &format!("p12 u={u}"), |_, _| {
            if !in_domain(u) {
                return f32::NAN;
            }
            fold_of(Monoid::SUM, 0, u as u64, |k| data[(k as usize).min(3)])
        });
    }
}
